    projectId: "iot-with-c-program-8460e",

    measurementId: "G-W2XEJ1HW0V"
  };

  const statusEl = document.getElementById("status");
  const connectionEl = document.getElementById("connection");

  function showError(message) {
    statusEl.textContent = message;
    connectionEl.textContent = "Not connected";
  }

  try {
    if (
      !firebaseConfig.databaseURL ||
      firebaseConfig.databaseURL === "YOUR_REALTIME_DATABASE_URL"
    ) {
      throw new Error(
        "Firebase Database URL missing. Add your actual Realtime Database URL."
      );
    }

    // INITIALIZE FIREBASE
    const app = initializeApp(firebaseConfig);

    // CONNECT TO REALTIME DATABASE
    const db = getDatabase(app);

    // READ DATA FROM /weather
    const weatherRef = ref(db, "weather");

    onValue(weatherRef, (snapshot) => {
      const data = snapshot.val();

      if (data === null) {
        statusEl.textContent =
          "Firebase connected. No data found at /weather.";
        connectionEl.textContent = "Connected";
        return;
      }

      document.getElementById("temperature").textContent =
        data.temperature ?? "--";

      document.getElementById("humidity").textContent =
        data.humidity ?? "--";

      document.getElementById("pressure").textContent =
        data.pressure ?? "--";

      statusEl.textContent = "✅ Firebase Connected — Live Data";
      connectionEl.textContent = "Connected";
      document.getElementById("lastUpdated").textContent =
        new Date().toLocaleString();

    }, (error) => {
      showError("Firebase read error: " + error.message);
    });

  } catch (error) {
    showError(error.message);
  }
